cmake_minimum_required(VERSION 3.16)
project(jetson_interfaces_practice LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

function(enable_strict_warnings target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endfunction()

add_executable(i2c_read src/i2c_read.cpp)
add_executable(uart_loopback src/uart_loopback.c)
add_executable(spi_loopback src/spi_loopback.c)
add_executable(gpio_v2_toggle src/gpio_v2_toggle.c)
add_executable(can_socketcan src/can_socketcan.cpp)
add_executable(pwm_sysfs src/pwm_sysfs.cpp)
find_package(Threads REQUIRED)
target_link_libraries(pwm_sysfs PRIVATE Threads::Threads)

foreach(target i2c_read uart_loopback spi_loopback gpio_v2_toggle can_socketcan pwm_sysfs)
  enable_strict_warnings(${target})
endforeach()

include(CTest)
if(BUILD_TESTING)
  foreach(target i2c_read uart_loopback spi_loopback gpio_v2_toggle can_socketcan pwm_sysfs)
    add_test(NAME ${target}_usage COMMAND $<TARGET_FILE:${target}>)
    set_tests_properties(${target}_usage PROPERTIES WILL_FAIL TRUE)
  endforeach()
endif()
